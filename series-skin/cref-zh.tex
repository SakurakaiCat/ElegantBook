% ============================================================================
% Chinese cross-references via cleveref
% hyperref 的 \autoref 默认输出 "section12.5" 等英文前缀.
% 用 cleveref 统一为中文习惯: 第12章 / 第12.5节 / 定理 12.32 / 式 (12-13).
% 再将 \autoref 映射到 \cref, 正文无需逐条改写.
% Load after hyperref (elegantbook loads hyperref itself).
% ============================================================================

\usepackage[nameinlink,noabbrev]{cleveref}

% --- 章节结构: 第…章 / 第…节 / 第…小节 ---
\crefformat{chapter}{第#2#1#3章}
\Crefformat{chapter}{第#2#1#3章}
\crefrangeformat{chapter}{第#3#1#4章至第#5#2#6章}
\Crefrangeformat{chapter}{第#3#1#4章至第#5#2#6章}

\crefformat{section}{第#2#1#3节}
\Crefformat{section}{第#2#1#3节}
\crefrangeformat{section}{第#3#1#4节至第#5#2#6节}
\Crefrangeformat{section}{第#3#1#4节至第#5#2#6节}

\crefformat{subsection}{第#2#1#3小节}
\Crefformat{subsection}{第#2#1#3小节}
\crefrangeformat{subsection}{第#3#1#4小节至第#5#2#6小节}
\Crefrangeformat{subsection}{第#3#1#4小节至第#5#2#6小节}

\crefformat{subsubsection}{第#2#1#3小节}
\Crefformat{subsubsection}{第#2#1#3小节}

\crefformat{paragraph}{第#2#1#3段}
\Crefformat{paragraph}{第#2#1#3段}

\crefformat{appendix}{附录~#2#1#3}
\Crefformat{appendix}{附录~#2#1#3}

% --- 图表 ---
\crefformat{figure}{图~#2#1#3}
\Crefformat{figure}{图~#2#1#3}
\crefformat{table}{表~#2#1#3}
\Crefformat{table}{表~#2#1#3}
\crefformat{subfigure}{图~#2#1#3}
\Crefformat{subfigure}{图~#2#1#3}

% --- 公式 ---
\crefformat{equation}{式~(#2#1#3)}
\Crefformat{equation}{式~(#2#1#3)}
\crefrangeformat{equation}{式~(#3#1#4)--(#5#2#6)}
\Crefrangeformat{equation}{式~(#3#1#4)--(#5#2#6)}

% --- 定理类 (与 elegantbook / amsthm 环境名对齐) ---
\crefname{theorem}{定理}{定理}
\Crefname{theorem}{定理}{定理}
\crefname{lemma}{引理}{引理}
\Crefname{lemma}{引理}{引理}
\crefname{corollary}{推论}{推论}
\Crefname{corollary}{推论}{推论}
\crefname{proposition}{命题}{命题}
\Crefname{proposition}{命题}{命题}
\crefname{definition}{定义}{定义}
\Crefname{definition}{定义}{定义}
\crefname{axiom}{公理}{公理}
\Crefname{axiom}{公理}{公理}
\crefname{postulate}{公设}{公设}
\Crefname{postulate}{公设}{公设}
\crefname{example}{例题}{例题}
\Crefname{example}{例题}{例题}
\crefname{exercise}{习题}{习题}
\Crefname{exercise}{习题}{习题}
\crefname{remark}{注}{注}
\Crefname{remark}{注}{注}

% 让正文既有的 \autoref{...} 自动走中文格式
\RenewDocumentCommand{\autoref}{s m}{%
  \IfBooleanTF{#1}{\cref*{#2}}{\cref{#2}}%
}
